# dotfetch Implementation Plan

> **For agentic workers:** REQUIRED SUB-SKILL: Use superpowers:subagent-driven-development (recommended) or superpowers:executing-plans to implement this plan task-by-task. Steps use checkbox (`- [ ]`) syntax for tracking.

**Goal:** Build a neofetch-style dotfiles status display (`dotfetch`) that shows custom ASCII art + system stats + dotfiles-specific fields, runnable as a standalone command and on terminal start.

**Architecture:** Single bash script (`scripts/dotfetch.sh`) with color setup, ASCII art, stat collector functions, a 2-column renderer, and a color swatch. A separate `scripts/audit.sh` handles cleanup analysis. The symlink module wires `dotfetch` into `~/.local/bin/`, and `zsh/05-integrations.zsh` calls it on terminal start.

**Tech Stack:** bash, standard POSIX tools (`uname`, `/etc/os-release`, `/proc/uptime`), git, existing `modules/_lib/platform.sh`

**Files:**
- Create: `scripts/dotfetch.sh`

- [ ] **Create the file with shebang, color setup, and ASCII art**

```bash
#!/usr/bin/env bash
# dotfetch — neofetch-style dotfiles info display
set -euo pipefail

DOTFETCH_DIR="$(cd "$(dirname "${BASH_SOURCE[0]}")/.." && pwd)"

# ── colors ────────────────────────────────────────────────────────────────────
_df_setup_colors() {
  if [[ -t 1 ]]; then
    C0='\e[0m'    # reset
    C1='\e[36m'   # cyan — accent (labels + ascii art)
    BOLD='\e[1m'
    DIM='\e[2m'
  else
    C0=''; C1=''; BOLD=''; DIM=''
  fi
}

# ── ascii art ─────────────────────────────────────────────────────────────────
# Width: 20 cols. Height: 12 lines.
_df_ascii() {
  ART=(
    "      /\\_____/\\"
    "     /  o   o  \\"
    "    ( ==  ^  == )"
    "     )         ("
    "    (           )"
    "     \\  |   |  /"
    "      \\_|___|_/"
    "                    "
    "  ╔════════════╗  "
    "  ║  BROWOLFF  ║  "
    "  ╚════════════╝  "
    "                    "
  )
}

# ── stat collectors ───────────────────────────────────────────────────────────

_df_get_os() {
  if [[ "$(uname -s)" == "Darwin" ]]; then
    local name version
    name="$(sw_vers -productName 2>/dev/null || echo macOS)"
    version="$(sw_vers -productVersion 2>/dev/null || echo '')"
    echo "${name} ${version}"
  elif [[ -f /etc/os-release ]]; then
    local name
    name="$(. /etc/os-release && echo "${PRETTY_NAME:-${NAME:-Linux}}")"
    echo "${name} $(uname -m)"
  else
    echo "$(uname -s) $(uname -m)"
  fi
}

_df_get_kernel() {
  uname -r
}

_df_get_uptime() {
  if [[ -f /proc/uptime ]]; then
    local secs
    secs="${EPOCHSECONDS:-}"
    if [[ -z "$secs" ]]; then
      secs="$(cut -d. -f1 /proc/uptime)"
    else
      secs="$(cut -d. -f1 /proc/uptime)"
    fi
    local days=$(( secs / 86400 ))
    local hours=$(( (secs % 86400) / 3600 ))
    local mins=$(( (secs % 3600) / 60 ))
    if (( days > 0 )); then echo "${days}d ${hours}h ${mins}m"
    elif (( hours > 0 )); then echo "${hours}h ${mins}m"
    else echo "${mins}m"
    fi
  elif [[ "$(uname -s)" == "Darwin" ]]; then
    local boot now secs
    boot="$(sysctl -n kern.boottime 2>/dev/null | awk '{print $4}' | tr -d ',')"
    now="$(date +%s)"
    secs=$(( now - boot ))
    local days=$(( secs / 86400 ))
    local hours=$(( (secs % 86400) / 3600 ))
    local mins=$(( (secs % 3600) / 60 ))
    if (( days > 0 )); then echo "${days}d ${hours}h ${mins}m"
    elif (( hours > 0 )); then echo "${hours}h ${mins}m"
    else echo "${mins}m"
    fi
  else
    echo "unknown"
  fi
}

_df_get_shell() {
  local shell_name
  shell_name="$(basename "${SHELL:-bash}")"
  case "$shell_name" in
    zsh)  echo "zsh ${ZSH_VERSION:-}";;
    bash) echo "bash ${BASH_VERSION:-}";;
    *)    echo "$shell_name";;
  esac
}

_df_get_terminal() {
  echo "${TERM_PROGRAM:-${TERM:-unknown}}"
}

_df_get_version() {
  local vfile="${DOTFETCH_DIR}/VERSION"
  [[ -f "$vfile" ]] && cat "$vfile" || echo "unknown"
}

_df_get_branch() {
  local branch dirty
  branch="$(git -C "$DOTFETCH_DIR" rev-parse --abbrev-ref HEAD 2>/dev/null || echo '?')"
  if git -C "$DOTFETCH_DIR" status --porcelain 2>/dev/null | grep -q .; then
    dirty=' ✗'
  else
    dirty=' ✓'
  fi
  echo "${branch}${dirty}"
}

_df_get_modules() {
  local total enabled disabled_in_conf
  total=0
  enabled=0
  for d in "${DOTFETCH_DIR}/modules"/*/; do
    [[ -f "${d}module.sh" ]] || continue
    (( total++ ))
  done
  # count !module lines in modules.conf
  local disabled=0
  if [[ -f "${DOTFETCH_DIR}/modules.conf" ]]; then
    disabled="$(grep -c '^!' "${DOTFETCH_DIR}/modules.conf" 2>/dev/null || echo 0)"
  fi
  enabled=$(( total - disabled ))
  echo "${enabled} / ${total} enabled"
}

_df_get_symlinks_fast() {
  # count create_symlink calls in symlinks.sh — no disk stat, instant
  local count
  count="$(grep -c 'create_symlink' "${DOTFETCH_DIR}/scripts/install/symlinks.sh" 2>/dev/null || echo '?')"
  echo "${count} defined"
}

_df_get_symlinks_full() {
  # actually walk targets and check
  local total=0 broken=0
  while IFS= read -r target; do
    [[ -L "$target" ]] || continue
    (( total++ ))
    [[ -e "$target" ]] || (( broken++ ))
  done < <(find "$HOME" -maxdepth 4 -type l 2>/dev/null)
  echo "${total} total  ${broken} broken"
}

# ── renderer ──────────────────────────────────────────────────────────────────

_df_render() {
  local -n _art=$1
  shift
  local -a stats=("$@")

  local art_width=22   # pad art lines to this width
  local label_color="${C1}"
  local reset="${C0}"

  local max=$(( ${#_art[@]} > ${#stats[@]} ? ${#_art[@]} : ${#stats[@]} ))

  for (( i=0; i<max; i++ )); do
    local art_line="${_art[$i]:-}"
    local stat_line="${stats[$i]:-}"

    # pad art line to fixed width for alignment
    local padded
    padded="$(printf "%-${art_width}s" "$art_line")"

    printf "${label_color}%s${reset}  %s\n" "$padded" "$stat_line"
  done
}

# ── color swatch ──────────────────────────────────────────────────────────────

_df_swatch() {
  [[ -t 1 ]] || return
  printf '\n%22s' ''  # indent to match art column
  for i in {0..7}; do printf "\e[4${i}m   \e[0m"; done
  printf '\n%22s' ''
  for i in {0..7}; do printf "\e[10${i}m   \e[0m"; done
  printf '\n\n'
}

# ── main ──────────────────────────────────────────────────────────────────────

main() {
  local mode="fast"
  for arg in "$@"; do
    case "$arg" in
      --full)  mode="full";;
      --audit) exec bash "${DOTFETCH_DIR}/scripts/audit.sh"; exit;;
      --help|-h)
        echo "Usage: dotfetch [--full] [--audit]"
        echo "  --full   include broken symlink scan (slower)"
        echo "  --audit  run cleanup audit"
        exit 0;;
    esac
  done

  _df_setup_colors
  _df_ascii

  local os kernel uptime shell terminal version branch modules symlinks
  os="$(_df_get_os)"
  kernel="$(_df_get_kernel)"
  uptime="$(_df_get_uptime)"
  shell="$(_df_get_shell)"
  terminal="$(_df_get_terminal)"
  version="$(_df_get_version)"
  branch="$(_df_get_branch)"
  modules="$(_df_get_modules)"
  if [[ "$mode" == "full" ]]; then
    symlinks="$(_df_get_symlinks_full)"
  else
    symlinks="$(_df_get_symlinks_fast)"
  fi

  local user_host="${BOLD}${USER:-$(whoami)}@${HOSTNAME:-$(hostname -s)}${C0}"
  local sep
  sep="${DIM}$(printf '%.0s─' {1..36})${C0}"

  local -a stats=(
    "$user_host"
    "$sep"
    "${C1}OS${C0}       ${os}"
    "${C1}Kernel${C0}   ${kernel}"
    "${C1}Uptime${C0}   ${uptime}"
    "${C1}Shell${C0}    ${shell}"
    "${C1}Terminal${C0} ${terminal}"
    ""
    "${DIM}── dotfiles $(printf '%.0s─' {1..18})${C0}"
    "${C1}Version${C0}  ${version}"
    "${C1}Branch${C0}   ${branch}"
    "${C1}Modules${C0}  ${modules}"
    "${C1}Symlinks${C0} ${symlinks}"
  )

  printf '\n'
  _df_render ART "${stats[@]}"
  _df_swatch
}

main "$@"
```

## Task 2: Create `scripts/audit.sh` — cleanup audit report

**Files:**
- Create: `scripts/audit.sh`

- [ ] **Create audit script**

```bash
#!/usr/bin/env bash
# audit.sh — dotfiles cleanup audit. Read-only. No auto-fix.
set -euo pipefail

DOTFILES_DIR="$(cd "$(dirname "${BASH_SOURCE[0]}")/.." && pwd)"

C0='\e[0m'
RED='\e[31m'
YEL='\e[33m'
CYN='\e[36m'
DIM='\e[2m'
BOLD='\e[1m'

[[ -t 1 ]] || { C0=''; RED=''; YEL=''; CYN=''; DIM=''; BOLD=''; }

_tag() { printf "${BOLD}[%-8s]${C0} %s\n" "$1" "$2"; }
_sep() { printf "${DIM}%.0s─${C0}" {1..60}; printf '\n'; }

printf "\n${BOLD}${CYN}dotfiles audit${C0} — $(date '+%Y-%m-%d %H:%M')\n"
_sep

# ── 1. stale module refs in modules.conf ─────────────────────────────────────
printf "\n${BOLD}1. modules.conf refs${C0}\n"
if [[ -f "${DOTFILES_DIR}/modules.conf" ]]; then
  while IFS= read -r line; do
    [[ "$line" =~ ^[[:space:]]*# ]] && continue
    [[ -z "$line" ]] && continue
    # strip leading ! or whitespace
    local_name="${line#\!}"
    local_name="${local_name#"${local_name%%[![:space:]]*}"}"
    if [[ ! -d "${DOTFILES_DIR}/modules/${local_name}" ]]; then
      _tag "stale" "modules.conf: '${line}' → no dir modules/${local_name}/"
    fi
  done < "${DOTFILES_DIR}/modules.conf"
  echo "  done"
else
  echo "  modules.conf not found"
fi

# ── 2. disabled modules that still exist (cleanup candidates) ─────────────────
printf "\n${BOLD}2. disabled modules (exist but opted-out)${C0}\n"
_found=0
if [[ -f "${DOTFILES_DIR}/modules.conf" ]]; then
  while IFS= read -r line; do
    [[ "$line" =~ ^\! ]] || continue
    local_name="${line#\!}"
    if [[ -d "${DOTFILES_DIR}/modules/${local_name}" ]]; then
      _tag "unused" "!${local_name} disabled in modules.conf but dir still present"
      _found=1
    fi
  done < "${DOTFILES_DIR}/modules.conf"
fi
(( _found == 0 )) && echo "  none"

# ── 3. zed-only code paths (zed is disabled via !zed) ─────────────────────────
printf "\n${BOLD}3. zed references (zed is disabled)${C0}\n"
_found=0
while IFS= read -r match; do
  # skip the zed module itself and audit.sh
  [[ "$match" == *"/modules/zed/"* ]] && continue
  [[ "$match" == *"/scripts/zed/"* ]] && continue
  [[ "$match" == *"/tui/tools/zed.sh"* ]] && continue
  [[ "$match" == *"audit.sh"* ]] && continue
  printf "${YEL}  %s${C0}\n" "$match"
  _found=1
done < <(grep -rn --include="*.sh" --include="*.zsh" --include="modules.conf" \
  -l 'zed' "${DOTFILES_DIR}" 2>/dev/null \
  | grep -v '\.git' \
  | grep -v '/modules/zed/' \
  | grep -v '/scripts/zed/' \
  | grep -v '/tui/tools/zed.sh' \
  | grep -v 'audit.sh')
(( _found == 0 )) && echo "  none"

# ── 4. dead symlinks on disk (targets missing) ────────────────────────────────
printf "\n${BOLD}4. broken symlinks on disk (${HOME})${C0}\n"
_found=0
while IFS= read -r link; do
  [[ -L "$link" && ! -e "$link" ]] || continue
  target="$(readlink "$link")"
  printf "${RED}  dead${C0}  %s → %s\n" "$link" "$target"
  _found=1
done < <(find "$HOME" -maxdepth 5 -type l 2>/dev/null)
(( _found == 0 )) && echo "  none"

# ── 5. orphan scripts (no callers in dotfiles) ────────────────────────────────
printf "\n${BOLD}5. potentially orphaned scripts in scripts/${C0}\n"
_found=0
while IFS= read -r script; do
  name="$(basename "$script")"
  # search for references to this script name anywhere in dotfiles (excluding itself)
  callers="$(grep -rl --include="*.sh" --include="*.zsh" --include="*.md" \
    "$name" "${DOTFILES_DIR}" 2>/dev/null \
    | grep -v "$script" | grep -v '\.git' || true)"
  if [[ -z "$callers" ]]; then
    _tag "unused" "scripts/${name##*scripts/} — no callers found"
    _found=1
  fi
done < <(find "${DOTFILES_DIR}/scripts" -name "*.sh" -not -path '*/.git/*' 2>/dev/null)
(( _found == 0 )) && echo "  none"

_sep
printf "${DIM}Audit complete. No changes made.${C0}\n\n"
```
