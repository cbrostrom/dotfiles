#!/bin/sh
set -eu
. /opt/linuxbro-samba/.env

fail() { printf '%s\n' "Samba preflight: $*" >&2; exit 1; }
[ "$(id -u)" = 0 ] || fail "Run as root"
[ "$PUID" = "$(id -u christian)" ] || fail "PUID differs from host christian"
[ "$PGID" = "$(id -g christian)" ] || fail "PGID differs from host christian"
[ "$PUID" -gt 0 ] || fail "Never use UID 0 for the share user"
[ "$PGID" -gt 0 ] || fail "Never use GID 0 for the share user"

# Use existing awk rather than introducing host Python.
awk -v address="$LAN_IP" -v network="$LAN_CIDR" '
function ipv4(value, octets, count, i, result) {
    count = split(value, octets, ".")
    if (count != 4) return -1
    result = 0
    for (i = 1; i <= 4; i++) {
        if (octets[i] !~ /^[0-9]+$/ || length(octets[i]) > 3 ||
            octets[i] + 0 > 255 ||
            (length(octets[i]) > 1 && substr(octets[i], 1, 1) == "0")) return -1
        result = result * 256 + octets[i]
    }
    return result
}
function private_ipv4(value) {
    return (value >= 167772160 && value <= 184549375) ||
           (value >= 2886729728 && value <= 2887778303) ||
           (value >= 3232235520 && value <= 3232301055)
}
BEGIN {
    count = split(network, cidr, "/")
    if (count != 2 || cidr[2] !~ /^[1-9][0-9]?$/ ||
        cidr[2] + 0 < 16 || cidr[2] + 0 > 32) exit 1
    ip_value = ipv4(address)
    net_value = ipv4(cidr[1])
    block_size = 2 ^ (32 - cidr[2])
    if (ip_value < 0 || net_value < 0 || net_value % block_size != 0) exit 1
    if (!private_ipv4(ip_value) || !private_ipv4(net_value) ||
        !private_ipv4(net_value + block_size - 1)) exit 1
    if (int(ip_value / block_size) != int(net_value / block_size)) exit 1
}' || fail "Use a valid RFC1918 LAN_IP and aligned LAN_CIDR (/16 to /32) containing it"
ip -4 -o address show dev "$LAN_IF" | awk '{print $4}' | cut -d/ -f1 |
    grep -Fxq "$LAN_IP" || fail "LAN_IP is not assigned to LAN_IF"

version=$(docker version --format '{{.Server.Version}}' | cut -d- -f1)
dpkg --compare-versions "$version" ge 28.0.0 || fail "Docker Engine 28+ required"
iptables -w -S DOCKER-USER >/dev/null || fail "DOCKER-USER unavailable; inspect backend"

actual_image=$(docker image inspect "$SAMBA_IMAGE" --format '{{.Id}}')
[ "$EXPECTED_IMAGE_ID" != REPLACE_ME ] || fail "Record the existing image ID"
[ "$actual_image" = "$EXPECTED_IMAGE_ID" ] || fail "Image changed; review before starting"
if docker inspect samba >/dev/null 2>&1; then
    [ "$(docker inspect samba --format '{{.State.Running}}')" = false ] ||
        fail "Old container samba is still running"
fi
test -f /etc/linuxbro-samba/smb.conf || fail "smb.conf missing"
test -d /home/christian/Dropbox/Musik || fail "Dropbox music folder missing"
test -d /media/4tb/Downloads || fail "Downloads folder missing"
test -s /var/lib/linuxbro-samba/private/passdb.tdb || fail "Migrated password database missing"
test -d /var/lib/linuxbro-samba/private || fail "private state directory missing"

check_disk() {
    mount_dir=$1
    expected=$2
    [ -n "$expected" ] && [ "$expected" != REPLACE_ME ] || fail "Set UUID for $mount_dir"
    mountpoint -q "$mount_dir" || fail "$mount_dir is not a mountpoint"
    actual=$(findmnt -rn --mountpoint "$mount_dir" --output UUID)
    [ "$actual" = "$expected" ] || fail "$mount_dir has wrong UUID: $actual"
    options=$(findmnt -rn --mountpoint "$mount_dir" --output OPTIONS)
    case ",$options," in
        *,rw,*) ;;
        *) fail "$mount_dir is not mounted read/write" ;;
    esac
}

check_disk /media/4tb "$DISK_4TB_UUID"

    "$SAMBA_IMAGE" --suppress-prompt /etc/samba/smb.conf >/dev/null
printf '%s\n' "Samba preflight passed"

Step 07

Create the script but do not apply it while the old server is running. It guards forwarded access to the dedicated bridge, including direct container-IP attempts, plus host input to the LAN-published port.

/usr/local/sbin/linuxbro-samba-firewall
#!/bin/sh
set -eu
. /opt/linuxbro-samba/.env
/usr/local/sbin/linuxbro-samba-preflight

iptables -w -S DOCKER-USER >/dev/null
iptables -w -N LBSMB-FWD 2>/dev/null || iptables -w -S LBSMB-FWD >/dev/null
iptables -w -N LBSMB-HOST 2>/dev/null || iptables -w -S LBSMB-HOST >/dev/null

# Rebuild only our chains. Apply while Samba is stopped, not during live traffic.
iptables -w -F LBSMB-FWD
iptables -w -A LBSMB-FWD -m conntrack --ctstate ESTABLISHED,RELATED -j RETURN
iptables -w -A LBSMB-FWD -i "$LAN_IF" -s "$LAN_CIDR" -p tcp \
    -m conntrack --ctdir ORIGINAL --ctorigdst "$LAN_IP" --ctorigdstport 445 -j RETURN
iptables -w -A LBSMB-FWD -j DROP

iptables -w -F LBSMB-HOST
iptables -w -A LBSMB-HOST -m conntrack --ctstate ESTABLISHED,RELATED -j RETURN
iptables -w -A LBSMB-HOST -i "$LAN_IF" -s "$LAN_CIDR" -j RETURN
iptables -w -A LBSMB-HOST -j DROP

# Put our checks first. Other services/ports are not matched.
while iptables -w -C DOCKER-USER -o br-samba -p tcp --dport 445 -j LBSMB-FWD 2>/dev/null; do
    iptables -w -D DOCKER-USER -o br-samba -p tcp --dport 445 -j LBSMB-FWD
done
iptables -w -I DOCKER-USER 1 -o br-samba -p tcp --dport 445 -j LBSMB-FWD

while iptables -w -C INPUT -p tcp -d "$LAN_IP" --dport 445 -j LBSMB-HOST 2>/dev/null; do
    iptables -w -D INPUT -p tcp -d "$LAN_IP" --dport 445 -j LBSMB-HOST
done
iptables -w -I INPUT 1 -p tcp -d "$LAN_IP" --dport 445 -j LBSMB-HOST
printf '%s\n' "Scoped Samba firewall installed"
Protect and syntax-check only
sudo chown root:root /usr/local/sbin/linuxbro-samba-firewall
sudo chmod 0755 /usr/local/sbin/linuxbro-samba-firewall
sudo sh -n /usr/local/sbin/linuxbro-samba-firewall

Verify the active backend first. If these chain names already belong to another configuration, stop rather than reuse them. Existing firewall managers may erase/reorder rules; retest after reload/reboot. No global flush or router/public forwarding is part of this guide.

#!/bin/sh
set -eu
umask 077
. /opt/linuxbro-samba/.env
fail() { printf '%s\n' "Migration: $*" >&2; exit 1; }
[ "$(id -u)" = 0 ] || fail "Run as root"
[ "$#" = 1 ] && [ "$1" = --stop-and-copy ] || fail "Use --stop-and-copy after reviewing the guide"
[ -f "$OLD_COMPOSE_FILE" ] || fail "Original Compose file unavailable; export it from the manager first"
[ -f "$OLD_APPDATA/config/smb.conf" ] || fail "Original configuration missing"
[ -s "$OLD_APPDATA/state/private/passdb.tdb" ] || fail "Original password database missing"
[ ! -e /var/lib/linuxbro-samba/private/passdb.tdb ] || fail "Destination database exists; do not overwrite"
[ "$(docker inspect samba --format '{{.State.Running}}')" = true ] || fail "Original container not running"
[ "$(docker inspect samba --format '{{.Image}}')" = "$EXPECTED_IMAGE_ID" ] || fail "Original image differs"
[ "$(docker image inspect "$SAMBA_IMAGE" --format '{{.Id}}')" = "$EXPECTED_IMAGE_ID" ] || fail "Image tag differs"

install -d -o root -g root -m 0700 /var/backups/linuxbro-samba
BACKUP_DIR=$(mktemp -d /var/backups/linuxbro-samba/migration.XXXXXX)
OLD_RESTART_POLICY=$(docker inspect samba --format \
    '{{.HostConfig.RestartPolicy.Name}}{{if eq .HostConfig.RestartPolicy.Name "on-failure"}}:{{.HostConfig.RestartPolicy.MaximumRetryCount}}{{end}}')
docker inspect samba > "$BACKUP_DIR/container-inspect.json"
iptables-save > "$BACKUP_DIR/iptables-before.txt"
tailscale serve status --json > "$BACKUP_DIR/tailscale-serve-before.json"
cp -a "$OLD_COMPOSE_FILE" "$BACKUP_DIR/original-compose.yaml"
old_dir=$(dirname "$OLD_COMPOSE_FILE")
for item in .env Dockerfile entrypoint.sh; do
    if [ -f "$old_dir/$item" ]; then
        cp -a "$old_dir/$item" "$BACKUP_DIR/original-$item"
    fi
done
printf 'BACKUP_DIR=%s\nOLD_RESTART_POLICY=%s\n' \
    "$BACKUP_DIR" "$OLD_RESTART_POLICY" > /etc/linuxbro-samba/migration.env
chmod 0600 /etc/linuxbro-samba/migration.env

changed=0
complete=0
trap '
    rc=$?
    trap - 0
    if [ "$complete" != 1 ] && [ "$changed" = 1 ]; then
        printf "%s\n" "Migration failed; attempting to restore the original container" >&2
        docker update --restart="$OLD_RESTART_POLICY" samba >/dev/null || true
        docker start samba >/dev/null || true
    fi
    exit "$rc"
' 0
docker update --restart=no samba >/dev/null
changed=1
docker stop --time 30 samba >/dev/null
[ "$(docker inspect samba --format '{{.State.Running}}')" = false ] || fail "Original container did not stop"
cp -a "$OLD_APPDATA/config/smb.conf" "$BACKUP_DIR/original-smb.conf"
cp -a "$OLD_APPDATA/state" "$BACKUP_DIR/state"
cp -a "$BACKUP_DIR/state/." /var/lib/linuxbro-samba/

The script records restart policy, metadata, Serve state, firewall snapshot and Compose files. Optional local .env/Dockerfile/entrypoint companions are backed up. Export any additional manager-owned overrides/dependencies yourself before proceeding; these cannot all be inferred from a label.

Backups are root-only because metadata/env/state can contain secrets. The script copies state only after stopping the daemon. On failure after changing restart policy it attempts to restore/start the old container; inspect the result rather than assuming recovery succeeded.

Checkpoint: original-stack automation paused; actual Compose accessible; backup/rollback understood; all files validated; new state location unoccupied; no unrelated unit/port/network conflict. Original server still running.

Step 11

Use your confirmed 100.100.1.100 tailnet address. Inspect existing Serve configuration before changing it. If TCP 445 belongs to an unrelated service, stop. If an existing Samba listener is being repointed, record its exact previous target for rollback.

Inspect first, then configure the private SMB proxy
sudo tailscale serve status
sudo tailscale serve --bg --tcp=445 tcp://127.0.0.1:1445
sudo tailscale serve status

Raw TCP, private tailnet only; --bg persists across restart. SMB provides required encryption. No tls-terminated-tcp, PROXY protocol or public Funnel.

Your chosen broad personal-tailnet policy is fine. Keep an intentional allow-all policy if only your devices belong. If policy is restrictive, merge access for your user/approved devices to TCP 445 on 100.100.1.100. Do not replace the entire policy. Grants/ACLs are additive, and Samba still requires your password.

The proxy hides original client IPs from Samba. Enforce tailnet membership through Tailscale, not broad source ranges in smb.conf. Protect your identity with MFA, revoke lost devices, and review access when sharing/inviting devices.

Updates are separate maintenance, not a migration rebuild

Use the existing reviewed build/update process only when security updates require it. Do not start the old Compose stack. Preserve an image reference, back up stopped state, validate refreshed packages/configuration, deliberately update EXPECTED_IMAGE_ID and repeat acceptance tests. Never roll back to a known-vulnerable image.

Debian's tracker currently needs review for Bookworm component issues; standalone/no clustering reduces feature applicability, not all future risk. Check package revisions regularly. The image-ID guard intentionally stops startup after a tag change until you review it.

Backups, monitoring and compromise

  • Back up protected project/config/scripts/unit/mount definitions.
  • Back up state while stopped or with a proven consistent method. Restrict/encrypt it because it contains credentials.
  • Keep offline/immutable data backups outside exported disks and inaccessible to Samba. Valid clients can ransomware writable files without a server exploit.
  • Test restoring sample data and state/identity with correct UID/GID/modes.
  • Alert through existing monitoring on unhealthy/stopped service, authentication anomalies, disk health/free space. No new monitoring stack required.

After a device/password compromise, revoke tailnet access, rotate through sudo docker exec -it linuxbro-samba smbpasswd christian, then restart to terminate sessions. Password rotation alone does not guarantee established sessions have ended.

Do not tune socket buffers/locking or disable security for throughput. Measure disk, LAN, tailnet direct/relay path and CPU limits. After fixing a startup cause, use sudo systemctl reset-failed linuxbro-samba before starting again.
